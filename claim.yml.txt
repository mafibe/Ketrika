name: Claim Faucet

on:
  workflow_dispatch:
    inputs:
      email:
        description: "Email du compte"
        required: true
        type: string
      platform:
        description: "Plateforme (ex: freetron, tronpick, litepick...)"
        required: true
        type: string
      userId:
        description: "USER_ID"
        required: true
        type: string

jobs:
  claim:
    runs-on: ubuntu-latest
    timeout-minutes: 8
    container:
      image: ghcr.io/aliya123-pro/camoufox-system:latest
      credentials:
        username: ${{ github.actor }}
        password: ${{ secrets.GITHUB_TOKEN }}

    steps:
      - name: Checkout du dépôt privé
        uses: actions/checkout@v4
        with:
          repository: Aliya123-pro/Aliyas
          token: ${{ secrets.PRIVATE_REPO_TOKEN }}
          path: private-code
          ref: main

      - name: Exécuter le claim
        working-directory: /app
        env:
          HOME: /home/camoxuser
          DISPLAY: ":99"
          CLAIM_EMAIL: ${{ github.event.inputs.email }}
          CLAIM_PLATFORM: ${{ github.event.inputs.platform }}
          USER_ID: ${{ github.event.inputs.userId }}
          GH_TOKEN: ${{ secrets.PRIVATE_REPO_TOKEN }}
          GH_USERNAME: Aliya123-pro
          GH_REPO: Aliyas
          GH_BRANCH: main
          JP_PROXY_LIST: ${{ secrets.JP_PROXY_LIST }}
          CRYPTO_SECRET: ${{ secrets.CRYPTO_SECRET }}
        run: |
          cp -r $GITHUB_WORKSPACE/private-code/* /app/
          mkdir -p /app/videos /app/screenshots
          xvfb-run --auto-servernum python voanjo.py

      - name: Upload des vidéos
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: claim-video-${{ github.run_id }}
          path: /app/videos/
          retention-days: 5
          if-no-files-found: warn
